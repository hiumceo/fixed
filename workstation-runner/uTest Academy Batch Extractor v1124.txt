  async function getJSON(url) {
    const response = await fetch(url, {
      credentials: "include"
    });

    if (!response.ok) {
      throw new Error(
        `${response.status} ${response.statusText}: ${url}`
      );
    }

    return response.json();
  }

  // ============================================================
  // TEXT UTILITIES
  // ============================================================

  function cleanText(value) {
    return String(value ?? "")
      .replace(/\u00a0/g, " ")
      .replace(/\s+/g, " ")
      .trim();
  }

  // ============================================================
  // INLINE SERIALIZER
  // ============================================================

  function serializeInline(node) {
    if (!node) {
      return [];
    }

    const output = [];

    function walk(element) {
      if (!element) {
        return;
      }

      if (
        element.nodeType ===
        Node.TEXT_NODE
      ) {
        if (element.nodeValue) {
          output.push({
            type: "text",
            text: element.nodeValue
          });
        }

        return;
      }

      if (
        element.nodeType !==
        Node.ELEMENT_NODE
      ) {
        return;
      }

      const tag =
        element.tagName.toLowerCase();

      // --------------------------------------------------------
      // LINE BREAK
      // --------------------------------------------------------

      if (tag === "br") {
        output.push({
          type: "break"
        });

        return;
      }

      // --------------------------------------------------------
      // LINK
      // --------------------------------------------------------

      if (tag === "a") {
        const children = [];

        for (
          const child
          of element.childNodes
        ) {
          const before =
            output.length;

          walk(child);

          children.push(
            ...output.splice(before)
          );
        }

        output.push({
          type: "link",
          text:
            cleanText(
              element.textContent
            ),
          url:
            element.href ||
            element.getAttribute(
              "href"
            ) ||
            "",
          content: children
        });

        return;
      }

      // --------------------------------------------------------
      // IMAGE
      // --------------------------------------------------------

      if (tag === "img") {
        output.push({
          type: "image",
          alt:
            element.getAttribute(
              "alt"
            ) || "",
          url:
            element.currentSrc ||
            element.src ||
            element.getAttribute(
              "src"
            ) ||
            ""
        });

        return;
      }

      // --------------------------------------------------------
      // FORMATTING
      // --------------------------------------------------------

      const formats = {
        strong: "strong",
        b: "b",
        em: "em",
        i: "i",
        u: "u",
        s: "s",
        del: "del",
        mark: "mark"
      };

      if (formats[tag]) {
        const children = [];

        for (
          const child
          of element.childNodes
        ) {
          const before =
            output.length;

          walk(child);

          children.push(
            ...output.splice(before)
          );
        }

        output.push({
          type: "format",
          format:
            formats[tag],
          content: children
        });

        return;
      }

      // --------------------------------------------------------
      // GENERIC INLINE ELEMENT
      // --------------------------------------------------------

      for (
        const child
        of element.childNodes
      ) {
        walk(child);
      }
    }

    walk(node);

    return output;
  }

  // ============================================================
  // TABLE CELL
  // ============================================================

  function serializeCell(cell) {
    return {
      tag:
        cell.tagName.toLowerCase(),

      colspan:
        Number(
          cell.getAttribute(
            "colspan"
          ) || 1
        ),

      rowspan:
        Number(
          cell.getAttribute(
            "rowspan"
          ) || 1
        ),

      content:
        serializeInline(cell)
    };
  }

  // ============================================================
  // TABLE
  // ============================================================

  function serializeTable(table) {
    const rows = [];

    const tableRows =
      table.querySelectorAll("tr");

    tableRows.forEach(row => {
      const cells =
        [...row.children]
          .filter(element =>
            element.tagName === "TH" ||
            element.tagName === "TD"
          );

      if (!cells.length) {
        return;
      }

      // Single TH spanning the table:
      // preserve as a semantic section/category.
      if (
        cells.length === 1 &&
        cells[0].tagName === "TH" &&
        Number(
          cells[0].getAttribute(
            "colspan"
          ) || 1
        ) > 1
      ) {
        rows.push({
          type: "section",

          colspan:
            Number(
              cells[0].getAttribute(
                "colspan"
              ) || 1
            ),

          content:
            serializeInline(
              cells[0]
            )
        });

        return;
      }

      rows.push({
        type: "row",

        cells:
          cells.map(
            serializeCell
          )
      });
    });

    return {
      type: "table",
      rows
    };
  }

  // ============================================================
  // LIST
  // ============================================================

  function serializeList(list) {
    const ordered =
      list.tagName.toLowerCase() ===
      "ol";

    const items = [];

    for (
      const li
      of [...list.children]
    ) {
      if (
        li.tagName !== "LI"
      ) {
        continue;
      }

      const content = [];
      const children = [];

      for (
        const child
        of li.childNodes
      ) {
        if (
          child.nodeType ===
          Node.ELEMENT_NODE
        ) {
          const tag =
            child.tagName.toLowerCase();

          if (
            tag === "ul" ||
            tag === "ol"
          ) {
            children.push(
              serializeList(child)
            );

            continue;
          }
        }

        const before =
          content.length;

        const inline =
          serializeInline(child);

        content.push(
          ...inline
        );
      }

      items.push({
        content,

        ...(children.length
          ? {
              children
            }
          : {})
      });
    }

    return {
      type: "list",
      ordered,
      items
    };
  }

  // ============================================================
  // MARKDOWN-ISH HEADING NORMALIZATION
  //
  // API content can contain:
  //
  // ### Heading
  //
  // while richer content is HTML.
  // Convert only Markdown heading lines.
  // ============================================================

  function normalizeMarkdownHeadings(
    source
  ) {
    if (!source) {
      return "";
    }

    return String(source).replace(
      /^\s*(#{1,6})\s+(.+?)\s*$/gm,
      (match, hashes, text) => {
        const level =
          hashes.length;

        return `<h${level}>${text}</h${level}>`;
      }
    );
  }

  // ============================================================
  // CONTENT SERIALIZER
  // ============================================================

  function serializeContent(source) {
    const normalized =
      normalizeMarkdownHeadings(
        source
      );

    const parser =
      new DOMParser();

    const doc =
      parser.parseFromString(
        normalized || "",
        "text/html"
      );

    // Never extract editor/page implementation noise.
    doc
      .querySelectorAll(
        "style,script,noscript,template,link,meta"
      )
      .forEach(element =>
        element.remove()
      );

    const content = [];

    function addNode(node) {
      if (!node) {
        return;
      }

      // --------------------------------------------------------
      // RAW TEXT
      // --------------------------------------------------------

      if (
        node.nodeType ===
        Node.TEXT_NODE
      ) {
        const text =
          cleanText(
            node.nodeValue
          );

        if (text) {
          content.push({
            type: "text",
            content: [
              {
                type: "text",
                text
              }
            ]
          });
        }

        return;
      }

      if (
        node.nodeType !==
        Node.ELEMENT_NODE
      ) {
        return;
      }

      const tag =
        node.tagName.toLowerCase();

      // --------------------------------------------------------
      // IGNORE
      // --------------------------------------------------------

      if (
        [
          "style",
          "script",
          "noscript",
          "template",
          "link",
          "meta"
        ].includes(tag)
      ) {
        return;
      }

      // --------------------------------------------------------
      // WISTIA
      // --------------------------------------------------------

      if (
        tag ===
        "wistia-player"
      ) {
        const mediaId =
          node.getAttribute(
            "media-id"
          );

        if (mediaId) {
          content.push({
            type: "video",
            provider: "wistia",
            mediaId
          });
        }

        return;
      }

      // --------------------------------------------------------
      // HEADING
      // --------------------------------------------------------

      if (
        /^h[1-6]$/.test(tag)
      ) {
        content.push({
          type: "heading",

          level:
            Number(
              tag.substring(1)
            ),

          content:
            serializeInline(node)
        });

        return;
      }

      // --------------------------------------------------------
      // PARAGRAPH
      // --------------------------------------------------------

      if (tag === "p") {
        const inline =
          serializeInline(node);

        if (inline.length) {
          content.push({
            type: "paragraph",
            content: inline
          });
        }

        return;
      }

      // --------------------------------------------------------
      // TABLE
      // --------------------------------------------------------

      if (tag === "table") {
        content.push(
          serializeTable(node)
        );

        return;
      }

      // --------------------------------------------------------
      // LIST
      // --------------------------------------------------------

      if (
        tag === "ul" ||
        tag === "ol"
      ) {
        content.push(
          serializeList(node)
        );

        return;
      }

      // --------------------------------------------------------
      // IMAGE
      // --------------------------------------------------------

      if (tag === "img") {
        content.push({
          type: "image",

          alt:
            node.getAttribute(
              "alt"
            ) || "",

          url:
            node.currentSrc ||
            node.src ||
            node.getAttribute(
              "src"
            ) ||
            ""
        });

        return;
      }

      // --------------------------------------------------------
      // BLOCKQUOTE
      // --------------------------------------------------------

      if (
        tag === "blockquote"
      ) {
        content.push({
          type: "blockquote",

          content:
            serializeInline(node)
        });

        return;
      }

      // --------------------------------------------------------
      // CODE
      // --------------------------------------------------------

      if (tag === "pre") {
        content.push({
          type: "code",

          content:
            node.textContent || ""
        });

        return;
      }

      // --------------------------------------------------------
      // SEPARATOR
      // --------------------------------------------------------

      if (tag === "hr") {
        content.push({
          type: "separator"
        });

        return;
      }

      // --------------------------------------------------------
      // GENERIC BLOCK CONTAINER
      // --------------------------------------------------------

      for (
        const child
        of node.childNodes
      ) {
        addNode(child);
      }
    }

    for (
      const child
      of doc.body.childNodes
    ) {
      addNode(child);
    }

    return content.filter(
      node => {
        if (
          node.type !==
          "text"
        ) {
          return true;
        }

        return node.content?.some(
          item =>
            item.type === "text" &&
            item.text.trim()
        );
      }
    );
  }

  // ============================================================
  // ZIP IMPLEMENTATION

  function crc32(bytes) {
    let crc =
      0xffffffff;

    for (
      let i = 0;
      i < bytes.length;
      i++
    ) {
      crc ^= bytes[i];

      for (
        let bit = 0;
        bit < 8;
        bit++
      ) {
        crc =
          (crc >>> 1) ^
          (
            -(crc & 1) &
            0xedb88320
          );
      }
    }

    return (
      crc ^
      0xffffffff
    ) >>> 0;
  }

  function u16(number) {
    return new Uint8Array([
      number & 255,
      (number >>> 8) & 255
    ]);
  }

  function u32(number) {
    return new Uint8Array([
      number & 255,
      (number >>> 8) & 255,
      (number >>> 16) & 255,
      (number >>> 24) & 255
    ]);
  }

  function concatArrays(
    ...arrays
  ) {
    const totalLength =
      arrays.reduce(
        (sum, array) =>
          sum + array.length,
        0
      );

    const result =
      new Uint8Array(
        totalLength
      );

    let offset = 0;

    for (
      const array
      of arrays
    ) {
      result.set(
        array,
        offset
      );

      offset +=
        array.length;
    }

    return result;
  }

  function createZip(files) {
    const encoder =
      new TextEncoder();

    const localParts = [];
    const centralParts = [];

    let offset = 0;

    for (
      const file
      of files
    ) {
      const nameBytes =
        encoder.encode(
          file.name
        );

      const dataBytes =
        encoder.encode(
          file.content
        );

      const checksum =
        crc32(
          dataBytes
        );

      // ZIP method 0 = stored/uncompressed.
      const localHeader =
        concatArrays(
          new Uint8Array([
            0x50, 0x4b,
            0x03, 0x04
          ]),

          u16(20), // version needed
          u16(0),  // flags
          u16(0),  // compression
          u16(0),  // mod time
          u16(0),  // mod date

          u32(checksum),

          u32(
            dataBytes.length
          ),

          u32(
            dataBytes.length
          ),

          u16(
            nameBytes.length
          ),

          u16(0),

          nameBytes
        );

      localParts.push(
        localHeader,
        dataBytes
      );

      const centralHeader =
        concatArrays(
          new Uint8Array([
            0x50, 0x4b,
            0x01, 0x02
          ]),

          u16(20), // made by
          u16(20), // needed

          u16(0),
          u16(0),
          u16(0),
          u16(0),

          u32(checksum),

          u32(
            dataBytes.length
          ),

          u32(
            dataBytes.length
          ),

          u16(
            nameBytes.length
          ),

          u16(0),
          u16(0),
          u16(0),
          u16(0),

          u32(0),

          u32(offset),

          nameBytes
        );

      centralParts.push(
        centralHeader
      );

      offset +=
        localHeader.length +
        dataBytes.length;
    }

    const local =
      concatArrays(
        ...localParts
      );

    const central =
      concatArrays(
        ...centralParts
      );

    const endRecord =
      concatArrays(
        new Uint8Array([
          0x50, 0x4b,
          0x05, 0x06
        ]),

        u16(0),
        u16(0),

        u16(files.length),
        u16(files.length),

        u32(
          central.length
        ),

        u32(
          local.length
        ),

        u16(0)
      );

    return new Blob(
      [
        local,
        central,
        endRecord
      ],
      {
        type:
          "application/zip"
      }
    );
  }

        file.name
      )
    ) {
      duplicateFilenames.push(
        file.name
      );
    }

    filenameSet.add(
      file.name
    );
  }

  if (
    duplicateFilenames.length
  ) {
    console.error(
      "DUPLICATE FILENAMES DETECTED:",
      duplicateFilenames
    );

    throw new Error(
      `ZIP aborted: ${duplicateFilenames.length} duplicate filename(s) detected.`
    );
  }

  console.log(
    "Filename verification: PASS"
  );

  console.log(
    `Unique ZIP filenames: ${filenameSet.size}`
  );

  // ------------------------------------------------------------
  // CREATE ZIP
  // ------------------------------------------------------------

  const zip =
    createZip(files);

  const downloadURL =
    URL.createObjectURL(
      zip
    );

    document.createElement(
      "a"
    );

  anchor.href =
    downloadURL;

  anchor.download =
    "uTest-Academy-v1124-JSON.zip";

  document.body.appendChild(
    anchor
  );

  anchor.click();

  anchor.remove();

  setTimeout(
    () =>
      URL.revokeObjectURL(
        downloadURL
      ),
    10000
  );
