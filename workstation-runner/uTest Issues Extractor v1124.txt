  async function fetchIssueHTML(
    issueURL
  ) {

    try {

        await fetch(
          issueURL,

            method:
              "GET",

            credentials:
              "include",

            redirect:
              "follow",

            headers: {
              accept:
                "text/html,application/xhtml+xml"
            }
          }
        );

        await response.text();

      const doc =
        new DOMParser()
          .parseFromString(
            html,
            "text/html"
          );

      return {

        url:
          issueURL,

        finalUrl:
          response.url,

        fetch: {

          method:
            "GET",

          status:
            response.status,

          ok:
            response.ok,

          redirected:
            response.redirected,

          contentType:
            response.headers.get(
              "content-type"
            ) || "",

          contentLength:
            html.length
        },

        content: {

          html,

          text:
            doc.body?.innerText ||
            ""
        }
      };

    } catch (error) {

      return {

        url:
          issueURL,

        finalUrl:
          null,

        fetch: {

          method:
            "GET",

          status:
            null,

          ok:
            false,

          redirected:
            false,

          contentType:
            "",

          contentLength:
            0
        },

        content: {

          html:
            "",

          text:
            ""
        },

        error:
          String(error)
      };
    }
  }

  // ==========================================================
  // FIND ATTACHMENTS INSIDE ANY API PAYLOAD
  //
  // We do not assume one exact nesting level.
  // ==========================================================

  function findAttachmentArrays(
    value,
    path = "$",
    results = []
  ) {

    if (
      value === null ||
      value === undefined
    ) {
      return results;
    }

    if (
      typeof value !== "object"
    ) {
      return results;
    }

    if (
      Array.isArray(value)
    ) {

      const looksLikeAttachments =
        value.length > 0 &&
        value.every(
          item =>
            item &&
            typeof item === "object" &&
            (
              "storageUrl" in item ||
              "fileName" in item ||
              "mimeType" in item ||
              "fileType" in item ||
              "canShareExternalLink" in item
            )
        );

      if (
        looksLikeAttachments
      ) {

        results.push({
          path,
          value
        });
      }

      for (
        let i = 0;
        i < value.length;
        i++
      ) {

        findAttachmentArrays(
          value[i],
          `${path}[${i}]`,
          results
        );
      }

      return results;
    }

    for (
      const [key, child]
      of Object.entries(value)
    ) {

      if (
        key.toLowerCase()
          .includes("attachment") &&
        Array.isArray(child)
      ) {

        results.push({
          path:
            `${path}.${key}`,

          value:
            child
        });
      }

      findAttachmentArrays(
        child,
        `${path}.${key}`,
        results
      );
    }

    return results;
  }

  function normalizeAttachments(
    arrays
  ) {

    const result = [];

    for (
      const entry
      of arrays
    ) {

      for (
        const attachment
        of entry.value
      ) {

        if (
          !attachment ||
          typeof attachment !==
            "object"
        ) {
          continue;
        }

        result.push({

          id:
            attachment.id ??
            null,

          storageUrl:
            attachment.storageUrl ??
            attachment.url ??
            null,

          fileName:
            attachment.fileName ??
            attachment.filename ??
            attachment.name ??
            null,

          fileSize:
            attachment.fileSize ??
            attachment.size ??
            null,

          mimeType:
            attachment.mimeType ??
            attachment.mime ??
            null,

          fileType:
            attachment.fileType ??
            null,

          canShareExternalLink:
            attachment.canShareExternalLink ??
            null,

          privateAttachment:
            attachment.privateAttachment ??
            null,

          sourcePath:
            entry.path,

          raw:
            attachment
        });
      }
    }

    return dedupeByKey(
      result
    );
  }

  // ==========================================================
  // RENDERED DOM FALLBACK
  //
  // Only used when API did not expose attachments.
  // ==========================================================

  function extractAttachmentsFromHTML(
    html
  ) {

    if (!html) {
      return [];
    }

    const doc =
      new DOMParser()
        .parseFromString(
          html,
          "text/html"
        );

    const result = [];

    const nodes =
      [
        ...doc.querySelectorAll(
          "ngf-attachment-preview"
        )
      ];

    for (
      const node
      of nodes
    ) {

      let attachment =
        null;

      const raw =
        node.getAttribute(
          "[attachment]"
        );

      if (raw) {

        try {

          attachment =
            JSON.parse(
              decodeHTMLAttribute(
                raw
              )
            );

        } catch {}
      }

      const link =
        node.querySelector(
          "a.file-name"
        );

      const media =
        node.querySelector(
          "video source, img"
        );

      const fileName =
        attachment?.fileName ||
        link?.getAttribute(
          "title"
        ) ||
        clean(
          link?.innerText ||
          ""
        ) ||
        null;

      const storageUrl =
        attachment?.storageUrl ||
        link?.href ||
        media?.getAttribute(
          "src"
        ) ||
        null;

      if (
        attachment?.id ||
        fileName ||
        storageUrl
      ) {

        result.push({

          id:
            attachment?.id ??
            null,

          storageUrl:
            storageUrl,

          fileName:
            fileName,

          fileSize:
            attachment?.fileSize ??
            null,

          mimeType:
            attachment?.mimeType ??
            null,

          fileType:
            attachment?.fileType ??
            null,

          canShareExternalLink:
            attachment?.canShareExternalLink ??
            null,

          source:
            "rendered-dom",

          html:
            node.outerHTML || "",

          raw:
            attachment
        });
      }
    }

    return dedupeByKey(
      result
    );
  }

          JSON.parse(raw);

          payload?.data

          ? payload.data

        response.ok &&
        Array.isArray(
          payload?.data

          ? "PASS"
          : "FAIL";

        url:
          `${location.origin}${url}`,

        fetch: {

            "GET",

          status:

          statusText:
            response.statusText,

            response.ok,

          redirected:
            response.redirected,

          contentType:
            response.headers.get(
              "content-type"
            ) || "",

          contentLength:
            raw.length
        },

        raw,

        payload,

        messages,

        count:
          messages.length,

        validation
      };

    } catch (error) {

      return {

        url:
          `${location.origin}${url}`,

        fetch: {

          method:
            "GET",

          status:
            null,

          statusText:
            null,

          ok:
            false
        },

        raw:
          "",

        payload:
          null,

        messages:
          [],

        count:
          0,

        validation:
          "FAIL",

        error:
          String(error)
      };
    }
  }
